CREATE OR REPLACE PROCEDURE SAJET.MAINTAIN_WO(TACTIVE IN VARCHAR2, TWO IN VARCHAR2, TRULE IN VARCHAR2, TPART IN VARCHAR2,
  TVERSION IN VARCHAR2, TFACTORYID IN VARCHAR2, TTARGET IN NUMBER, TROUTE IN VARCHAR2, TPDLINE IN VARCHAR2,
  TSTART IN VARCHAR2, TEND IN VARCHAR2, TSCHEDULE IN DATE, TDUE IN DATE, TEMPID IN VARCHAR2, 
  TTYPE IN VARCHAR2, TREMARK IN VARCHAR2, TRES OUT VARCHAR2, TFIELD OUT VARCHAR2,   
  TCUSTOMER_CODE IN VARCHAR2, TBURNIN_TIME IN VARCHAR2, TPO_NO IN VARCHAR2, 
  TSALES_ORDER IN VARCHAR2, TMASTER_WO IN VARCHAR2) IS
C_PART SAJET.SYS_PART.PART_ID%TYPE;
C_ROUTE SAJET.SYS_ROUTE.ROUTE_ID%TYPE;
C_START SAJET.SYS_PROCESS.PROCESS_ID%TYPE;
C_END SAJET.SYS_PROCESS.PROCESS_ID%TYPE;
C_LINE SAJET.SYS_PDLINE.PDLINE_ID%TYPE;
C_ROUTE1 SAJET.G_WO_BASE.ROUTE_ID%TYPE;
C_CUST SAJET.SYS_CUSTOMER.CUSTOMER_ID%TYPE;  
BEGIN
  TRES := 'OK';
  TFIELD := 'PART';
  SELECT PART_ID INTO C_PART FROM SAJET.SYS_PART WHERE PART_NO = TPART AND ROWNUM = 1;
  TFIELD := 'ROUTE';
  SELECT ROUTE_ID INTO C_ROUTE FROM SAJET.SYS_ROUTE WHERE ROUTE_NAME = TROUTE AND ROWNUM = 1;
  TFIELD := 'START PROCESS';  
  SELECT PROCESS_ID INTO C_START FROM SAJET.SYS_PROCESS WHERE PROCESS_NAME = TSTART AND ROWNUM = 1;
  TFIELD := 'END PROCESS';
  SELECT PROCESS_ID INTO C_END FROM SAJET.SYS_PROCESS WHERE PROCESS_NAME = TEND AND ROWNUM = 1;
  IF TPDLINE IS NOT NULL THEN
    TFIELD := 'PDLINE';
    SELECT PDLINE_ID INTO C_LINE FROM SAJET.SYS_PDLINE WHERE PDLINE_NAME = TPDLINE AND ROWNUM = 1;
  END IF;  
  IF TCUSTOMER_CODE IS NOT NULL THEN
    TFIELD := 'CUSTOMER';
    SELECT CUSTOMER_ID INTO C_CUST FROM SAJET.SYS_CUSTOMER WHERE CUSTOMER_CODE = TCUSTOMER_CODE AND ROWNUM = 1;
  END IF;  
  IF TACTIVE = 'APPEND' THEN
    INSERT INTO SAJET.G_WO_BASE (WORK_ORDER, WO_TYPE, WO_RULE, PART_ID, VERSION, FACTORY_ID, WO_CREATE_DATE, 
      WO_SCHEDULE_DATE,WO_DUE_DATE,ROUTE_ID,TARGET_QTY,WO_STATUS,DEFAULT_PDLINE_ID,START_PROCESS_ID,END_PROCESS_ID,CUSTOMER_ID,
      PO_NO,SALES_ORDER,MASTER_WO,REMARK,BURNIN_TIME,UPDATE_USERID,UPDATE_TIME)
      VALUES(TWO, TTYPE, TRULE, C_PART, TVERSION, TFACTORYID, SYSDATE, 
      TSCHEDULE, TDUE, C_ROUTE, TTARGET, 1, C_LINE, C_START, C_END, C_CUST, 
      TPO_NO, TSALES_ORDER, TMASTER_WO, TREMARK, TBURNIN_TIME, TEMPID, SYSDATE);   
  ELSIF TACTIVE = 'MODIFY' THEN
    TFIELD := 'WO';
    SELECT ROUTE_ID INTO C_ROUTE1 FROM SAJET.G_WO_BASE WHERE WORK_ORDER = TWO AND ROWNUM = 1;
    UPDATE SAJET.G_WO_BASE SET WO_TYPE = TTYPE, WO_RULE = TRULE, PART_ID = C_PART, VERSION = TVERSION 
      ,WO_SCHEDULE_DATE = TSCHEDULE, WO_DUE_DATE = TDUE, ROUTE_ID = C_ROUTE, TARGET_QTY = TTARGET               
      ,DEFAULT_PDLINE_ID = C_LINE, START_PROCESS_ID = C_START, END_PROCESS_ID = C_END, REMARK = TREMARK
      ,UPDATE_USERID = TEMPID, UPDATE_TIME = SYSDATE 
      ,CUSTOMER_ID = C_CUST, PO_NO = TPO_NO, SALES_ORDER = TSALES_ORDER, MASTER_WO = TMASTER_WO
      ,BURNIN_TIME = TBURNIN_TIME
      WHERE WORK_ORDER = TWO AND ROWNUM = 1;
    IF C_ROUTE1 <> C_ROUTE THEN
      UPDATE SAJET.G_SN_STATUS SET ROUTE_ID = C_ROUTE 
        WHERE WORK_ORDER = TWO AND ROUTE_ID = C_ROUTE1 AND PROCESS_ID = 0;
    END IF;  
  END IF;
  INSERT INTO SAJET.G_HT_WO_BASE SELECT * FROM SAJET.G_WO_BASE WHERE WORK_ORDER = TWO AND ROWNUM = 1;
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    TRES := 'NOT FOUND';
  WHEN OTHERS THEN
    TRES := SQLERRM;
END;

GRANT EXECUTE ON SAJET.MAINTAIN_WO TO SYS_USER;
